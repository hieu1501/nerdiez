    function formatDate(value) {
      const date = new Date(value);
      return Number.isNaN(date.getTime()) ? '' : date.toLocaleDateString('en', {
        month: 'short',
        day: 'numeric',
        year: 'numeric'
      });

      return articles

      const vote = localVotes.get(article.__backendId);
      card.querySelectorAll('[data-vote]').forEach(button => {
        button.classList.toggle('voted-up', vote === 'up' && button.dataset.vote === 'up');

        button.setAttribute('aria-pressed', String(vote === button.dataset.vote));

      card.querySelector('[data-field="downvotes"]').textContent = article.downvotes || 0;

    }

    function closeReader() {
      reader.classList.remove('open');
      reader.setAttribute('aria-hidden', 'true');
      currentArticleId = null;
    }

    async function vote(id, direction) {
      const article = articles.find(item => item.__backendId === id);
      if (!article) return;

      const oldVote = localVotes.get(id);
      if (oldVote === direction) {
        document.getElementById('status-message').textContent = 'Your vote is already counted for this article.';
        return;
      }

      const controls = [
        ...document.querySelectorAll(`[data-article-id="${id}"] [data-vote]`),
        ...document.querySelectorAll('#reader-upvote, #reader-downvote')
      ];

      controls.forEach(control => control.disabled = true);

      const updated = { ...article };
      if (oldVote === 'up') updated.upvotes = Math.max(0, (updated.upvotes || 0) - 1);
      if (oldVote === 'down') updated.downvotes = Math.max(0, (updated.downvotes || 0) - 1);
      if (direction === 'up') updated.upvotes = (updated.upvotes || 0) + 1;
      if (direction === 'down') updated.downvotes = (updated.downvotes || 0) + 1;

      localVotes.set(id, direction);
      document.getElementById('status-message').textContent = 'Saving your vote…';

      const result = await window.dataSdk.update(updated);

      if (!result.isOk) {
        if (oldVote) localVotes.set(id, oldVote);
        else localVotes.delete(id);
        document.getElementById('status-message').textContent = 'Your vote could not be saved. Please try again.';
      } else {
        document.getElementById('status-message').textContent = 'Thanks — your vote was saved.';
      }

      controls.forEach(control => control.disabled = false);
      renderArticles();

      if (currentArticleId === id) {
        const refreshedArticle = articles.find(item => item.__backendId === id);
        if (refreshedArticle) openReader(id);
      }
    }

    function toggleTheme() {
      const darkMode = document.body.classList.toggle('dark-mode');
      const button = document.getElementById('theme-toggle');
      button.setAttribute('aria-pressed', String(darkMode));
      button.querySelector('.theme-icon-light').classList.toggle('hidden', darkMode);
      button.querySelector('.theme-icon-dark').classList.toggle('hidden', !darkMode);
    }

    document.getElementById('articles-nav-button').addEventListener('click', () => {
      document.getElementById('articles').scrollIntoView({ behavior: 'smooth', block: 'start' });
    });

    document.getElementById('profile-nav-button').addEventListener('click', () => {
      const panel = document.getElementById('profile-panel');
      const isOpen = panel.classList.toggle('open');
      document.getElementById('profile-nav-button').setAttribute('aria-expanded', String(isOpen));
    });

    document.getElementById('theme-toggle').addEventListener('click', toggleTheme);

    document.querySelectorAll('.filter-btn').forEach(button => {
      button.addEventListener('click', () => {
        activeTopic = button.dataset.topic;
        activeSubtopic = 'All';
        document.querySelectorAll('.filter-btn').forEach(item => {
          item.setAttribute('aria-pressed', String(item === button));
        });
        renderSubtopics();
        renderArticles();
      });
    });

    document.getElementById('close-reader').addEventListener('click', closeReader);
    document.getElementById('reader-backdrop').addEventListener('click', closeReader);
    document.getElementById('reader-upvote').addEventListener('click', () => currentArticleId && vote(currentArticleId, 'up'));
    document.getElementById('reader-downvote').addEventListener('click', () => currentArticleId && vote(currentArticleId, 'down'));

    document.addEventListener('keydown', event => {
      if (event.key === 'Escape') {
        closeReader();
        document.getElementById('profile-panel').classList.remove('open');
        document.getElementById('profile-nav-button').setAttribute('aria-expanded', 'false');
      }
    });

    const dataHandler = {
      onDataChanged(data) {
        if (data.length) articles = data;
        renderSubtopics();
        renderArticles();

        if (currentArticleId) {
          const current = articles.find(article => article.__backendId === currentArticleId);
          if (current) openReader(currentArticleId);
          else closeReader();
        }
      }
    };

    async function initialize() {
      const result = await window.dataSdk.init(dataHandler);
      if (!result.isOk) {
        document.getElementById('status-message').textContent = 'Articles are temporarily unavailable. Please refresh and try again.';
      }
    }

    lucide.createIcons();
    initialize();
  </script>
 </body>
